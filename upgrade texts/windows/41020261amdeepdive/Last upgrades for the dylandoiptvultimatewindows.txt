## 🔴 FIX 1 — LIVE TV CATEGORY SCROLLING

The categories aren't scrollable because the container is missing overflow and height constraints.

```typescript
// pages/LiveTV.tsx — Complete layout with scrollable category sidebar

export const LiveTVPage: React.FC<LiveTVProps> = ({
  channels, categories, epgData, getNowPlaying,
  playerState, onChannelSelect, settings, favorites, onToggleFavorite
}) => {
  const [selectedCategoryId, setSelectedCategoryId] = useState<string | null>(null);
  const [searchQuery, setSearchQuery] = useState('');
  const [view, setView] = useState<'split' | 'epg'>('split');
  const categoryScrollRef = useRef<HTMLDivElement>(null);

  const filteredChannels = useMemo(() => {
    let result = channels;
    if (selectedCategoryId) {
      result = result.filter(c => c.category_id === selectedCategoryId);
    }
    if (searchQuery.trim()) {
      const q = searchQuery.toLowerCase();
      result = result.filter(c => c.name?.toLowerCase().includes(q));
    }
    return result;
  }, [channels, selectedCategoryId, searchQuery]);

  return (
    <div className="flex h-full overflow-hidden">

      {/* ══ LEFT: Category Sidebar — FULLY SCROLLABLE ══════════════════ */}
      <div className="w-44 shrink-0 flex flex-col border-r border-white/8 overflow-hidden">
        {/* Header */}
        <div className="px-3 py-2.5 border-b border-white/8 shrink-0">
          <p className="text-[10px] font-bold uppercase tracking-widest text-white/30">
            Categories
          </p>
        </div>

        {/* ── THE FIX: flex-1 + overflow-y-auto makes it scroll ── */}
        <div
          ref={categoryScrollRef}
          className="flex-1 overflow-y-auto py-1
            scrollbar-thin scrollbar-thumb-white/10 scrollbar-track-transparent
            hover:scrollbar-thumb-white/20"
        >
          {/* ALL option */}
          <button
            onClick={() => setSelectedCategoryId(null)}
            className={`
              w-full text-left px-3 py-2.5 text-sm transition-all border-r-2 flex items-center justify-between
              ${!selectedCategoryId
                ? 'bg-cyan-500/10 text-cyan-400 border-cyan-400 font-medium'
                : 'text-white/50 border-transparent hover:text-white hover:bg-white/5'
              }
            `}
          >
            <span className="truncate">All Channels</span>
            <span className="text-[10px] ml-1 shrink-0 opacity-50">
              {channels.length}
            </span>
          </button>

          {/* Category items — all scrollable */}
          {categories.map(cat => {
            const count = channels.filter(c => c.category_id === cat.category_id).length;
            return (
              <button
                key={cat.category_id}
                onClick={() => setSelectedCategoryId(cat.category_id)}
                className={`
                  w-full text-left px-3 py-2.5 text-sm transition-all border-r-2
                  flex items-center justify-between group
                  ${selectedCategoryId === cat.category_id
                    ? 'bg-cyan-500/10 text-cyan-400 border-cyan-400 font-medium'
                    : 'text-white/50 border-transparent hover:text-white hover:bg-white/5'
                  }
                `}
              >
                <span className="truncate text-xs">{cat.category_name}</span>
                <span className="text-[9px] ml-1 shrink-0 opacity-40">{count}</span>
              </button>
            );
          })}
        </div>

        {/* View toggle at bottom */}
        <div className="border-t border-white/8 p-2 shrink-0 flex gap-1">
          <button
            onClick={() => setView('split')}
            className={`flex-1 py-1.5 rounded-lg text-[10px] font-semibold transition-all ${
              view === 'split'
                ? 'bg-cyan-500/20 text-cyan-400'
                : 'text-white/30 hover:text-white/60'
            }`}
          >
            List
          </button>
          <button
            onClick={() => setView('epg')}
            className={`flex-1 py-1.5 rounded-lg text-[10px] font-semibold transition-all ${
              view === 'epg'
                ? 'bg-cyan-500/20 text-cyan-400'
                : 'text-white/30 hover:text-white/60'
            }`}
          >
            Guide
          </button>
        </div>
      </div>

      {/* ══ CENTER/RIGHT: Content ═══════════════════════════════════════ */}
      <div className="flex-1 flex overflow-hidden">

          <SplitView
            channels={filteredChannels}
            searchQuery={searchQuery}
            onSearchChange={setSearchQuery}
            playerState={playerState}
            onChannelSelect={onChannelSelect}
            getNowPlaying={getNowPlaying}

            epgData={epgData}

            onChannelClick={ch => { onChannelSelect(ch); setView('split'); }}
```

## 🟣 FEATURE 1 — MULTI-PROVIDER MANAGEMENT SYSTEM

```typescript
// components/ProviderManager.tsx — Complete multi-provider UI

import React, { useState, useCallback } from 'react';
import {
  Plus, Tv2, Trash2, Edit2, Check, Wifi, WifiOff,
  Star, ChevronRight, RefreshCw, AlertCircle, Shield
} from 'lucide-react';
import { XtreamProfile } from '../types/xtream';

interface ProviderManagerProps {
  providers: XtreamProfile[];
  activeProviderId: string | null;
  onActivate: (provider: XtreamProfile) => void;
  onAdd: (data: ProviderFormData) => Promise<void>;
  onEdit: (id: string, data: ProviderFormData) => Promise<void>;
  onDelete: (id: string) => void;
  onRefresh: (id: string) => Promise<void>;
  isOpen: boolean;
  onClose: () => void;
}

interface ProviderFormData {
  name: string;
  server: string;
  username: string;
  password: string;
  color: string; // accent color per provider
}

interface ProviderStatus {
  [id: string]: 'connected' | 'error' | 'testing' | 'idle';
}

const PROVIDER_COLORS = [
  '#00FFFF', '#FF00FF', '#FF6B35', '#00FF88',
  '#FFD700', '#FF3366', '#7B61FF', '#00D4FF'
];

const emptyForm = (): ProviderFormData => ({
  name: '', server: '', username: '', password: '', color: PROVIDER_COLORS[0]
});

export const ProviderManager: React.FC<ProviderManagerProps> = ({
  providers, activeProviderId, onActivate, onAdd, onEdit,
  onDelete, onRefresh, isOpen, onClose
}) => {
  const [mode, setMode] = useState<'list' | 'add' | 'edit'>('list');
  const [editingId, setEditingId] = useState<string | null>(null);
  const [form, setForm] = useState<ProviderFormData>(emptyForm());
  const [status, setStatus] = useState<ProviderStatus>({});
  const [isSaving, setIsSaving] = useState(false);
  const [showPassword, setShowPassword] = useState(false);
  const [testResult, setTestResult] = useState<string | null>(null);

  if (!isOpen) return null;

  const handleTest = async () => {
    if (!form.server || !form.username || !form.password) return;
    setTestResult('testing');
    try {
      const url = `${form.server.replace(/\/$/, '')}/player_api.php?username=${form.username}&password=${form.password}`;
      const res = await fetch(url, { signal: AbortSignal.timeout(10000) });
      const data = await res.json();
      if (data?.user_info?.auth === 1) {
        setTestResult('success');
      } else {
        setTestResult('failed');
      }
    } catch {
      setTestResult('error');
    }
  };

  const handleSave = async () => {
    if (!form.server || !form.username || !form.password) return;
    setIsSaving(true);
    try {
      if (mode === 'edit' && editingId) {
        await onEdit(editingId, form);
      } else {
        await onAdd(form);
      }
      setMode('list');
      setForm(emptyForm());
      setTestResult(null);
    } finally {
      setIsSaving(false);
    }
  };

  const startEdit = (provider: XtreamProfile) => {
    setEditingId(provider.id);
    setForm({
      name: provider.name || '',
      server: provider.server,
      username: provider.username,
      password: provider.password,
      color: provider.color || PROVIDER_COLORS[0],
    });
    setTestResult(null);
    setMode('edit');
  };

  return (
    <div
      className="fixed inset-0 z-[200] flex items-center justify-center bg-black/80 backdrop-blur-sm"
      onClick={e => { if (e.target === e.currentTarget) onClose(); }}
    >
      <div className="w-[540px] max-h-[85vh] flex flex-col bg-[#080810] 
        border border-white/10 rounded-2xl shadow-2xl overflow-hidden">

        {/* ── Header ───────────────────────────────────────────────── */}
        <div className="flex items-center justify-between px-6 py-4 
          border-b border-white/8 shrink-0">
          <div>
            <h2 className="text-white text-base font-bold">IPTV Providers</h2>
            <p className="text-white/40 text-xs mt-0.5">
              {providers.length} provider{providers.length !== 1 ? 's' : ''} configured
            </p>
          </div>
          <div className="flex items-center gap-2">
            {mode === 'list' && (
              <button
                onClick={() => { setMode('add'); setForm(emptyForm()); setTestResult(null); }}
                className="flex items-center gap-1.5 px-3 py-1.5 
                  bg-cyan-500/15 text-cyan-400 border border-cyan-500/30 
                  rounded-xl text-xs font-semibold hover:bg-cyan-500/25 transition-all"
              >
                <Plus size={13} /> Add Provider
              </button>
            )}
            <button
              onClick={onClose}
              className="text-white/30 hover:text-white transition-colors text-lg leading-none"
            >
              ×
            </button>
          </div>
        </div>

        {/* ── Provider List ─────────────────────────────────────────── */}
        {mode === 'list' && (
          <div className="flex-1 overflow-y-auto divide-y divide-white/5">
            {providers.length === 0 ? (
              <div className="flex flex-col items-center justify-center py-16 gap-3">
                <Tv2 size={40} className="text-white/15" />
                <p className="text-white/30 text-sm">No providers yet</p>
                <button
                  onClick={() => setMode('add')}
                  className="px-4 py-2 bg-cyan-500/15 text-cyan-400 border border-cyan-500/30 
                    rounded-xl text-xs font-semibold"
                >
                  Add Your First Provider
                </button>
              </div>
            ) : (
              providers.map(provider => {
                const isActive = provider.id === activeProviderId;
                const provStatus = status[provider.id] ?? 'idle';
                const accentColor = provider.color || '#00FFFF';

                return (
                  <div
                    key={provider.id}
                    className={`
                      flex items-center gap-4 px-5 py-4 transition-all cursor-pointer
                      ${isActive ? 'bg-white/4' : 'hover:bg-white/3'}
                    `}
                    onClick={() => onActivate(provider)}
                  >
                    {/* Color dot + active indicator */}
                    <div className="relative shrink-0">
                      <div
                        className="w-10 h-10 rounded-xl flex items-center justify-center"
                        style={{ backgroundColor: `${accentColor}20`, border: `1px solid ${accentColor}30` }}
                      >
                        <Tv2 size={18} style={{ color: accentColor }} />
                      </div>
                      {isActive && (
                        <div
                          className="absolute -top-1 -right-1 w-4 h-4 rounded-full 
                            flex items-center justify-center"
                          style={{ backgroundColor: accentColor }}
                        >
                          <Check size={9} className="text-black" strokeWidth={3} />
                        </div>
                      )}
                    </div>

                    {/* Info */}
                    <div className="flex-1 min-w-0">
                      <div className="flex items-center gap-2">
                        <p className="text-white font-semibold text-sm truncate">
                          {provider.name || provider.username}
                        </p>
                        {isActive && (
                          <span
                            className="text-[9px] px-1.5 py-0.5 rounded font-bold"
                            style={{ color: accentColor, backgroundColor: `${accentColor}20` }}
                          >
                            ACTIVE
                          </span>
                        )}
                      </div>
                      <p className="text-white/35 text-xs truncate mt-0.5">{provider.server}</p>
                      <p className="text-white/25 text-[10px] truncate">@{provider.username}</p>
                    </div>

                    {/* Status */}
                    <div className="flex items-center gap-1 shrink-0">
                      {provStatus === 'testing' && (
                        <RefreshCw size={13} className="text-yellow-400 animate-spin" />
                      )}
                      {provStatus === 'connected' && (
                        <Wifi size={13} className="text-green-400" />
                      )}
                      {provStatus === 'error' && (
                        <WifiOff size={13} className="text-red-400" />
                      )}
                    </div>

                    {/* Actions */}
                    <div className="flex items-center gap-1 shrink-0"
                         onClick={e => e.stopPropagation()}>
                      <button
                        onClick={() => startEdit(provider)}
                        className="p-1.5 text-white/25 hover:text-cyan-400 transition-colors rounded-lg hover:bg-white/5"
                        title="Edit"
                      >
                        <Edit2 size={13} />
                      </button>
                      <button
                        onClick={async () => {
                          setStatus(p => ({ ...p, [provider.id]: 'testing' }));
                          await onRefresh(provider.id);
                          setStatus(p => ({ ...p, [provider.id]: 'connected' }));
                        }}
                        className="p-1.5 text-white/25 hover:text-green-400 transition-colors rounded-lg hover:bg-white/5"
                        title="Refresh"
                      >
                        <RefreshCw size={13} />
                      </button>
                      <button
                        onClick={() => {
                          if (confirm(`Remove "${provider.name || provider.username}"?`)) {
                            onDelete(provider.id);
                          }
                        }}
                        className="p-1.5 text-white/25 hover:text-red-400 transition-colors rounded-lg hover:bg-white/5"
                        title="Delete"
                      >
                        <Trash2 size={13} />
                      </button>
                    </div>
                  </div>
                );
              })
            )}
          </div>
        )}

        {/* ── Add / Edit Form ───────────────────────────────────────── */}
        {(mode === 'add' || mode === 'edit') && (
          <div className="flex-1 overflow-y-auto px-6 py-5 space-y-4">
            {/* Provider name */}
            <FormField label="Provider Name" hint="A friendly label for this provider">
              <input
                value={form.name}
                onChange={e => setForm(p => ({ ...p, name: e.target.value }))}
                placeholder="e.g. Main IPTV, Sports Package, Family Plan"

              />

            <FormField label="Server URL" hint="Include port number if required">

              />

                <input

                    value={form.password}
                    onChange={e => setForm(p => ({ ...p, password: e.target.value }))}
                    placeholder="password"
                    type={showPassword ? 'text' : 'password'}
                    className={`${inputClass} pr-9`}

                      backgroundColor: color,
```

```typescript
// components/EpgGuide.tsx — Full cable-style EPG grid

import React, {

import { ChevronLeft, ChevronRight, Radio, Calendar, Clock } from 'lucide-react';
import { XtreamChannel } from '../types/xtream';
import { EPGProgram } from '../types/epg';

interface EpgGuideProps {
  channels: XtreamChannel[];
  epgData: Map<string, EPGProgram[]>;
  getNowPlaying: (channelId: string) => EPGProgram | null;
  onChannelClick: (channel: XtreamChannel) => void;

const CH_W = 180;       // channel column width px
const HOUR_W = 240;     // pixels per 1 hour
const ROW_H = 64;       // row height px
const HEADER_H = 48;    // time header height px
const VISIBLE_ROWS = 10; // virtualize rows outside this range

const fmt = (date: Date, format: '12h' | '24h') =>
  date.toLocaleTimeString([], {
    hour: '2-digit', minute: '2-digit',
    hour12: format === '12h'
  });

const fmtDuration = (startTime: string, stopTime: string): string => {
  const mins = Math.round(
    (new Date(stopTime).getTime() - new Date(startTime).getTime()) / 60000
  );
  return mins >= 60
    ? `${Math.floor(mins / 60)}h ${mins % 60}m`
    : `${mins}m`;
};

  channels, epgData, getNowPlaying, onChannelClick,

  const headerRef = useRef<HTMLDivElement>(null);
  const channelRef = useRef<HTMLDivElement>(null);

    program: EPGProgram; channel: XtreamChannel; x: number; y: number
  } | null>(null);
  const [currentTime, setCurrentTime] = useState(new Date());
  const [scrollLeft, setScrollLeft] = useState(0);

  // Update clock every 30 seconds
  useEffect(() => {
    const interval = setInterval(() => setCurrentTime(new Date()), 30000);
    return () => clearInterval(interval);
  }, []);

  // Day start = today at midnight
  const dayStart = useMemo(() => {
    const d = new Date(currentTime);
    d.setHours(0, 0, 0, 0);
    return d;
  }, [currentTime.toDateString()]);

  const nowX = useMemo(() => {
    const elapsed = currentTime.getTime() - dayStart.getTime();
    return (elapsed / 3600000) * HOUR_W;
  }, [currentTime, dayStart]);

  // Scroll to NOW on mount
  useEffect(() => {
    if (!bodyRef.current) return;
    const targetX = Math.max(0, nowX - CH_W - 60);

    if (headerRef.current) headerRef.current.scrollLeft = targetX;

    const sl = (e.target as HTMLDivElement).scrollLeft;

    if (headerRef.current) headerRef.current.scrollLeft = sl;
    if (channelRef.current) channelRef.current.scrollTop = st;
```

```typescript
import React, { useEffect, useRef } from 'react';
import { AppSettings } from '../types/app';

interface SubtitleRendererProps {
  videoRef: React.RefObject<HTMLVideoElement>;
  settings: AppSettings;
  externalSubtitleUrl?: string;
  isVisible: boolean;
}

export const SubtitleRenderer: React.FC<SubtitleRendererProps> = ({
  videoRef, settings, externalSubtitleUrl, isVisible
}) => {
  const [currentCue, setCurrentCue] = React.useState<string>('');
  const trackRef = useRef<HTMLTrackElement | null>(null);

  // Apply subtitle styles via CSS variables
  useEffect(() => {
    const root = document.documentElement;
    root.style.setProperty('--sub-font-family', settings.subtitleFontFamily ?? 'Arial');
    root.style.setProperty('--sub-font-size', `${settings.subtitleFontSize ?? 22}px`);
    root.style.setProperty('--sub-color', settings.subtitleColor ?? '#FFFFFF');
    root.style.setProperty('--sub-outline', settings.subtitleOutlineColor ?? '#000000');
    root.style.setProperty('--sub-outline-w', `${settings.subtitleOutlineWidth ?? 2}px`);
    root.style.setProperty('--sub-bg-opacity', `${settings.subtitleBackgroundOpacity ?? 0}`);
  }, [
    settings.subtitleFontFamily,
    settings.subtitleFontSize,
    settings.subtitleColor,
    settings.subtitleOutlineColor,
    settings.subtitleOutlineWidth,
    settings.subtitleBackgroundOpacity,
  ]);

  // Listen to text track cue changes
  useEffect(() => {
    const video = videoRef.current;
    if (!video) return;

    const updateCue = () => {
      const tracks = video.textTracks;
      for (let i = 0; i < tracks.length; i++) {
        const track = tracks[i];
        if (track.mode === 'showing' || track.mode === 'hidden') {
          const activeCues = track.activeCues;
          if (activeCues && activeCues.length > 0) {
            const cue = activeCues[0] as VTTCue;
            setCurrentCue(cue.text || '');
            return;
          }
        }
      }
      setCurrentCue('');
    };

    const tracks = video.textTracks;
    for (let i = 0; i < tracks.length; i++) {
      tracks[i].addEventListener('cuechange', updateCue);
      if (settings.subtitlesEnabled) {
        tracks[i].mode = 'hidden'; // We render manually
      }
    }

    return () => {
      for (let i = 0; i < tracks.length; i++) {
        tracks[i].removeEventListener('cuechange', updateCue);
      }
    };
  }, [videoRef, settings.subtitlesEnabled]);

  if (!isVisible || !settings.subtitlesEnabled || !currentCue) return null;

  const lines = currentCue.split('\n').filter(Boolean);

  return (
    <div className="absolute bottom-16 left-0 right-0 flex flex-col items-center 
      gap-1 px-8 pointer-events-none z-20">
      {lines.map((line, i) => (
        <span
          key={i}
          className="inline-block max-w-[85%] text-center"
          style={{
            fontFamily: `var(--sub-font-family)`,
            fontSize: `var(--sub-font-size)`,
            color: `var(--sub-color)`,
            WebkitTextStroke: `var(--sub-outline-w) var(--sub-outline)`,
            textShadow: `0 0 4px var(--sub-outline), 0 0 8px var(--sub-outline)`,
            backgroundColor: `rgba(0,0,0,var(--sub-bg-opacity))`,
            padding: '2px 10px',
            borderRadius: '4px',
            lineHeight: '1.4',
            letterSpacing: '0.01em',
          }}
          dangerouslySetInnerHTML={{ __html: line }}
        />
      ))}
    </div>
  );
};
```

```javascript
// electron/main.cjs — Complete DVR system with FFmpeg

const { spawn, exec } = require('child_process');
const path = require('path');
const fs = require('fs').promises;
const fsSync = require('fs');

// ── DVR State ──────────────────────────────────────────────────────────────
const MAX_CONCURRENT_RECORDINGS = 3;
const recordings = new Map(); // recordingId → { process, meta, startTime, outputPath }
let dvrOutputDir = null;

const getDvrDir = () => {
  if (dvrOutputDir) return dvrOutputDir;
  dvrOutputDir = path.join(app.getPath('videos'), 'DYLANDOS IPTV DVR');
  fsSync.mkdirSync(dvrOutputDir, { recursive: true });
  return dvrOutputDir;
};

// ── Find FFmpeg ────────────────────────────────────────────────────────────
const findFFmpeg = () => {
  // 1. Check bundled ffmpeg in app resources
  const bundledPaths = [
    path.join(process.resourcesPath, 'ffmpeg', 'ffmpeg.exe'),
    path.join(__dirname, '..', 'resources', 'ffmpeg', 'ffmpeg.exe'),
  ];
  for (const p of bundledPaths) {
    if (fsSync.existsSync(p)) return p;
  }
  // 2. Check PATH
  return 'ffmpeg';
};

const FFMPEG_PATH = findFFmpeg();

// ── IPC: Start Recording ───────────────────────────────────────────────────
ipcMain.handle('dvr:start', async (_e, {
  streamUrl,
  channelName,
  programTitle,
  durationSeconds, // optional — record for N seconds then auto-stop
  customOutputDir,
}) => {
  if (recordings.size >= MAX_CONCURRENT_RECORDINGS) {
    return {
      success: false,
      error: `Maximum ${MAX_CONCURRENT_RECORDINGS} concurrent recordings reached.`,
    };
  }

  const recordingId = `rec_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`;
  const outputDir = customOutputDir || getDvrDir();

  // Safe filename
  const safeChannelName = channelName.replace(/[^a-zA-Z0-9\-_\s]/g, '').trim().slice(0, 50);
  const safeProgramTitle = (programTitle || 'Recording')
    .replace(/[^a-zA-Z0-9\-_\s]/g, '').trim().slice(0, 80);
  const timestamp = new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19);
  const filename = `${safeChannelName} - ${safeProgramTitle} - ${timestamp}.ts`;
  const outputPath = path.join(outputDir, filename);

  // FFmpeg args for recording
  const args = [
    '-hide_banner',
    '-loglevel', 'error',
    '-user_agent', 'Mozilla/5.0',
    '-i', streamUrl,
    '-c', 'copy',           // Stream copy — no re-encode = no CPU overhead
    '-f', 'mpegts',         // Output as MPEG-TS for universal compatibility
    '-avoid_negative_ts', 'make_zero',
  ];

  if (durationSeconds && durationSeconds > 0) {
    args.push('-t', String(durationSeconds));
  }

  args.push(outputPath);

  const meta = {
    id: recordingId,
    channelName,
    programTitle: programTitle || 'Unknown Program',
    streamUrl,
    outputPath,
    filename,
    startTime: new Date().toISOString(),
    durationSeconds: durationSeconds || null,
    status: 'recording',
    size: 0,
  };

  try {
    const ffmpegProcess = spawn(FFMPEG_PATH, args, {
      stdio: ['ignore', 'pipe', 'pipe'],
    });

    recordings.set(recordingId, { process: ffmpegProcess, meta, startTime: Date.now() });

    // Size tracking interval
    const sizeInterval = setInterval(async () => {
      try {
        const stat = await fs.stat(outputPath);
        meta.size = stat.size;
        // Push progress to renderer
        win?.webContents.send('dvr:progress', {
          id: recordingId,
          size: stat.size,
          elapsedSeconds: Math.round((Date.now() - recordings.get(recordingId).startTime) / 1000),
          status: 'recording',
        });
      } catch { /* file not created yet */ }
    }, 2000);

    // On process exit
    ffmpegProcess.on('close', async (code) => {
      clearInterval(sizeInterval);
      try {
        const stat = await fs.stat(outputPath);
        meta.size = stat.size;
        meta.status = code === 0 ? 'completed' : 'error';
      } catch {
        meta.status = 'error';
      }

      recordings.delete(recordingId);
      await saveRecordingToLibrary(meta);

      win?.webContents.send('dvr:completed', {
        id: recordingId,
        status: meta.status,
        outputPath,
        filename,
        size: meta.size,
      });
    });

    ffmpegProcess.stderr.on('data', (data) => {
      const line = data.toString();
      if (line.includes('Error') || line.includes('Invalid')) {
        console.error(`[DVR ${recordingId}] FFmpeg: ${line}`);
      }
    });

    await saveRecordingToLibrary(meta); // Save initial record
    return { success: true, recordingId, outputPath };

  } catch (err) {
    return { success: false, error: err.message };
  }
});

// ── IPC: Stop Recording ────────────────────────────────────────────────────
ipcMain.handle('dvr:stop', async (_e, recordingId) => {
  const recording = recordings.get(recordingId);
  if (!recording) return { success: false, error: 'Recording not found' };

  recording.process.stdin?.write('q'); // Graceful FFmpeg quit
  setTimeout(() => {
    if (!recording.process.killed) {
      recording.process.kill('SIGTERM'); // Force if needed
    }
  }, 3000);

  return { success: true };
});

// ── IPC: Stop All Recordings ───────────────────────────────────────────────
ipcMain.handle('dvr:stop-all', async () => {
  for (const [id, rec] of recordings.entries()) {
    rec.process.stdin?.write('q');
    setTimeout(() => rec.process.kill('SIGTERM'), 2000);
  }
  return { success: true, stoppedCount: recordings.size };
});

// ── IPC: List Active Recordings ────────────────────────────────────────────
ipcMain.handle('dvr:list-active', () => {
  const active = [];
  for (const [id, rec] of recordings.entries()) {
    active.push({
      ...rec.meta,
      elapsedSeconds: Math.round((Date.now() - rec.startTime) / 1000),
    });
  }
  return { recordings: active, maxConcurrent: MAX_CONCURRENT_RECORDINGS };
});

// ── IPC: List Recording Library ────────────────────────────────────────────
ipcMain.handle('dvr:list-library', async (_e, dirOverride) => {
  const dir = dirOverride || getDvrDir();
  try {
    const libraryFile = path.join(app.getPath('userData'), 'data', 'dvr-library.json');
    const raw = await fs.readFile(libraryFile, 'utf8');
    return JSON.parse(raw);
  } catch {
    return [];
  }
});

// ── IPC: Delete Recording ──────────────────────────────────────────────────
ipcMain.handle('dvr:delete', async (_e, { recordingId, deleteFile }) => {
  try {
    const libraryFile = path.join(app.getPath('userData'), 'data', 'dvr-library.json');
    const library = JSON.parse(await fs.readFile(libraryFile, 'utf8').catch(() => '[]'));
    const entry = library.find(r => r.id === recordingId);

    if (entry && deleteFile) {
      await fs.unlink(entry.outputPath).catch(() => {});
    }

    const updated = library.filter(r => r.id !== recordingId);
    await fs.writeFile(libraryFile, JSON.stringify(updated, null, 2));
    return { success: true };
  } catch (err) {
    return { success: false, error: err.message };
  }
});

// ── IPC: Set DVR Output Directory ──────────────────────────────────────────
ipcMain.handle('dvr:set-output-dir', (_e, dir) => {
  dvrOutputDir = dir;
  fsSync.mkdirSync(dir, { recursive: true });
});

// ── IPC: Get DVR Status ────────────────────────────────────────────────────
ipcMain.handle('dvr:status', () => ({
  activeCount: recordings.size,
  maxConcurrent: MAX_CONCURRENT_RECORDINGS,
  ffmpegAvailable: fsSync.existsSync(FFMPEG_PATH) || FFMPEG_PATH === 'ffmpeg',
  outputDir: getDvrDir(),
}));

// ── IPC: Check FFmpeg availability ─────────────────────────────────────────
ipcMain.handle('dvr:check-ffmpeg', () => {
  return new Promise(resolve => {
    exec(`${FFMPEG_PATH} -version`, (error) => {
      resolve({ available: !error, path: FFMPEG_PATH });
    });
  });
});

// ── Helper: Save to library JSON ───────────────────────────────────────────
async function saveRecordingToLibrary(meta) {
  const libraryFile = path.join(app.getPath('userData'), 'data', 'dvr-library.json');
  let library = [];
  try {
    library = JSON.parse(await fs.readFile(libraryFile, 'utf8'));
  } catch { /* new file */ }

  const idx = library.findIndex(r => r.id === meta.id);
  if (idx >= 0) {
    library[idx] = meta;
  } else {
    library.unshift(meta);
  }

  await fs.mkdir(path.dirname(libraryFile), { recursive: true });
  await fs.writeFile(libraryFile, JSON.stringify(library, null, 2));
}
```

```typescript
      <div className="flex gap-0 px-6 pt-3 border-b border-white/8 shrink-0">
        {[
          { id: 'active', label: 'Active', icon: <Radio size={13} />, count: activeRecordings.length },
          { id: 'library', label: 'Library', icon: <Film size={13} />, count: library.filter(r => r.status === 'completed').length },
          { id: 'schedule', label: 'Scheduled', icon: <Clock size={13} />, count: 0 },
        ].map(t => (
          <button
            key={t.id}
            onClick={() => setTab(t.id as typeof tab)}
            className={`
              flex items-center gap-1.5 px-4 py-2.5 text-xs font-semibold
              border-b-2 transition-all -mb-px
              ${tab === t.id
                ? 'border-current text-white'
                : 'border-transparent text-white/35 hover:text-white/60'
              }
            `}
            style={tab === t.id ? { borderColor: accentColor, color: 'white' } : {}}
          >
            {t.icon}
            {t.label}
            {t.count > 0 && (
              <span
                className="text-[9px] px-1.5 py-0.5 rounded-full font-bold"
                style={tab === t.id
                  ? { background: `${accentColor}30`, color: accentColor }
                  : { background: 'rgba(255,255,255,0.1)', color: 'rgba(255,255,255,0.4)' }
                }
              >
                {t.count}
              </span>
            )}
          </button>
        ))}
      </div>

      {/* ── Content ──────────────────────────────────────────────────── */}
      <div className="flex-1 overflow-y-auto">

        {/* Active Recordings */}
        {tab === 'active' && (
          <div className="p-6 space-y-3">
            {activeRecordings.length === 0 ? (
              <EmptyDVRState accentColor={accentColor} onRecord={() => setShowNewRecording(true)} />
            ) : (
              activeRecordings.map(rec => (
                <ActiveRecordingCard
                  key={rec.id}
                  recording={rec}
                  onStop={() => handleStopRecording(rec.id)}
                  accentColor={accentColor}
                />
              ))
            )}

            {/* Slot guide */}
            {activeRecordings.length > 0 && (
              <div className="mt-4 p-4 rounded-xl border border-white/6 bg-white/2">
                <p className="text-white/40 text-xs text-center">
                  {(dvrStatus?.maxConcurrent ?? 3) - activeRecordings.length} recording slot{
                    (dvrStatus?.maxConcurrent ?? 3) - activeRecordings.length !== 1 ? 's' : ''
                  } available · You can watch any channel while recording
                </p>
              </div>
            )}
          </div>
        )}

        {/* Library */}
        {tab === 'library' && (
          <div className="p-6">
            {library.filter(r => r.status === 'completed').length === 0 ? (
              <div className="text-center py-16">
                <Film size={40} className="mx-auto text-white/15 mb-3" />
                <p className="text-white/30 text-sm">No recordings yet</p>
              </div>
            ) : (

                      key={rec.id}
                      recording={rec}
                      onPlay={() => onPlayFile(rec.outputPath)}
                      onDelete={(deleteFile) => handleDeleteFromLibrary(rec.id, deleteFile)}
                      onOpenFolder={() => window.electronAPI.openExternal(
                        `file://${rec.outputPath.substring(0, rec.outputPath.lastIndexOf('\\'))}`
                      )}
                      accentColor={accentColor}
                    />
                  ))}
              </div>
            )}
          </div>
        )}

        {/* Scheduled */}
        {tab === 'schedule' && (
          <div className="flex flex-col items-center justify-center h-64 gap-3">
            <Clock size={40} className="text-white/15" />
            <p className="text-white/30 text-sm">Scheduled recordings coming soon</p>
            <p className="text-white/20 text-xs">Set a recording timer from the EPG guide</p>
          </div>
        )}
      </div>

      {/* ── New Recording Dialog ──────────────────────────────────────── */}
      {showNewRecording && (
        <NewRecordingDialog
          channels={channels}
          onClose={() => setShowNewRecording(false)}
          onStart={async (data) => {
            const result = await window.electronAPI.invoke('dvr:start', data);
            if (result.success) {
              setShowNewRecording(false);
              const { recordings } = await window.electronAPI.invoke('dvr:list-active');
              setActiveRecordings(recordings);
            }
            return result;
          }}
          accentColor={accentColor}
          maxSlots={dvrStatus?.maxConcurrent ?? 3}
          activeCount={dvrStatus?.activeCount ?? 0}
        />
      )}
    </div>

// New Recording Dialog
const NewRecordingDialog: React.FC<{
  channels: XtreamChannel[];
  onClose: () => void;
  onStart: (data: unknown) => Promise<{ success: boolean; error?: string }>;
  accentColor: string;
  maxSlots: number;
  activeCount: number;
}> = ({ channels, onClose, onStart, accentColor, maxSlots, activeCount }) => {
  const [search, setSearch] = useState('');
  const [selectedChannel, setSelectedChannel] = useState<XtreamChannel | null>(null);
  const [programTitle, setProgramTitle] = useState('');
  const [duration, setDuration] = useState(60); // minutes
  const [useTimer, setUseTimer] = useState(false);
  const [isStarting, setIsStarting] = useState(false);
  const [error, setError] = useState('');

  const filteredChannels = useMemo(() =>
    channels.filter(c => c.name?.toLowerCase().includes(search.toLowerCase())).slice(0, 50),
    [channels, search]
  );

  const handleStart = async () => {
    if (!selectedChannel) return;
    setIsStarting(true);
    setError('');
    const result = await onStart({
      streamUrl: `${selectedChannel.stream_url}`,
      channelName: selectedChannel.name,
      programTitle: programTitle || selectedChannel.name,
      durationSeconds: useTimer ? duration * 60 : null,
    });
    if (!result.success) {
      setError(result.error || 'Failed to start recording');
      setIsStarting(false);
    }
  };

  return (
    <div className="fixed inset-0 z-[300] flex items-center justify-center bg-black/80 backdrop-blur-sm">
      <div className="w-[480px] max-h-[85vh] flex flex-col bg-[#080810] border border-white/10 rounded-2xl shadow-2xl overflow-hidden">

        <div className="px-5 py-4 border-b border-white/8 shrink-0">
          <h3 className="text-white font-bold text-base flex items-center gap-2">
            <Circle size={14} className="text-red-400 fill-red-400 animate-pulse" />
            New Recording
          </h3>
          <p className="text-white/40 text-xs mt-0.5">
            Slot {activeCount + 1} of {maxSlots}
          </p>
        </div>

        <div className="flex-1 overflow-y-auto p-5 space-y-4">
          {/* Channel search */}
          <div>
            <label className="text-white/50 text-xs font-semibold block mb-1.5">Channel</label>
            <input
              value={search}
              onChange={e => setSearch(e.target.value)}
              placeholder="Search channels..."
              className="w-full bg-white/5 border border-white/10 rounded-xl px-3 py-2.5 
                text-white text-sm placeholder:text-white/20 focus:outline-none focus:border-cyan-500/40"
            />
            <div className="mt-2 max-h-36 overflow-y-auto rounded-xl border border-white/8 divide-y divide-white/5">
              {filteredChannels.map(ch => (
                <button
                  key={ch.stream_id}
                  onClick={() => { setSelectedChannel(ch); setProgramTitle(ch.name); }}
                  className={`w-full flex items-center gap-2 px-3 py-2.5 text-left text-xs transition-colors ${
                    selectedChannel?.stream_id === ch.stream_id
                      ? 'bg-white/8 text-white'
                      : 'text-white/60 hover:bg-white/5 hover:text-white'
                  }`}
                >
                  <div className="w-1.5 h-1.5 rounded-full shrink-0"
                       style={{ background: selectedChannel?.stream_id === ch.stream_id ? accentColor : 'rgba(255,255,255,0.2)' }} />
                  {ch.name}
                </button>
              ))}
            </div>
          </div>

          {/* Program title */}
          <div>
            <label className="text-white/50 text-xs font-semibold block mb-1.5">Recording Title</label>
            <input
              value={programTitle}
              onChange={e => setProgramTitle(e.target.value)}
              placeholder="Recording title..."
              className="w-full bg-white/5 border border-white/10 rounded-xl px-3 py-2.5 
                text-white text-sm placeholder:text-white/20 focus:outline-none focus:border-cyan-500/40"
            />
          </div>

          {/* Duration timer */}
          <div className="flex items-center justify-between p-3 rounded-xl border border-white/8 bg-white/3">
            <div>
              <p className="text-white/80 text-sm font-medium">Auto-Stop Timer</p>
              <p className="text-white/35 text-xs">Stop recording automatically after set time</p>
            </div>
            <button
              onClick={() => setUseTimer(p => !p)}
              className={`relative w-11 h-6 rounded-full transition-colors ${useTimer ? '' : 'bg-white/15'}`}
              style={useTimer ? { background: accentColor } : {}}
            >
              <div className={`absolute top-0.5 left-0.5 w-5 h-5 bg-white rounded-full shadow transition-transform ${useTimer ? 'translate-x-5' : ''}`} />
            </button>
          </div>

          {useTimer && (
            <div>
              <label className="text-white/50 text-xs font-semibold block mb-1.5">
                Duration — {duration} minutes ({Math.floor(duration / 60)}h {duration % 60}m)
              </label>
              <input
                type="range" min={5} max={480} step={5} value={duration}
                onChange={e => setDuration(parseInt(e.target.value))}
                className="w-full" style={{ accentColor }}
              />
              <div className="flex justify-between text-[10px] text-white/25 mt-1">
                <span>5m</span><span>1h</span><span>2h</span><span>4h</span><span>8h</span>
              </div>
            </div>
          )}

          {error && (
            <div className="flex items-center gap-2 px-3 py-2.5 rounded-xl 
              bg-red-500/10 border border-red-500/30 text-red-400 text-xs">
              <AlertCircle size={13} /> {error}
            </div>
          )}
        </div>

        <div className="px-5 py-4 border-t border-white/8 flex gap-2 shrink-0">
          <button onClick={onClose} className="px-4 py-2.5 text-white/50 text-sm hover:text-white transition-colors">
            Cancel
          </button>
          <button
            onClick={handleStart}
            disabled={!selectedChannel || isStarting}
            className="flex-1 flex items-center justify-center gap-2 py-2.5 text-sm font-bold
              rounded-xl transition-all disabled:opacity-40 disabled:cursor-not-allowed"
            style={{ background: 'linear-gradient(135deg, #ef4444, #dc2626)', color: 'white' }}
          >
            <Circle size={13} className="fill-white animate-pulse" />
            {isStarting ? 'Starting...' : 'Start Recording'}
          </button>
        </div>
      </div>
    </div>
  );
};

);
```

```typescript
export const applyTheme = (theme: Theme): void => {

  root.style.setProperty('--theme-bg1', theme.bg1);
  root.style.setProperty('--theme-bg2', theme.bg2);
  root.style.setProperty('--theme-bg3', theme.bg3);
  root.style.setProperty('--theme-bg-glass', theme.bgGlass);
  root.style.setProperty('--theme-accent1', theme.accent1);
  root.style.setProperty('--theme-accent2', theme.accent2);
  root.style.setProperty('--theme-accent-live', theme.accentLive);
  root.style.setProperty('--theme-text-primary', theme.textPrimary);
  root.style.setProperty('--theme-text-secondary', theme.textSecondary);
  root.style.setProperty('--theme-text-muted', theme.textMuted);
  root.style.setProperty('--theme-border', theme.border);
  root.style.setProperty('--theme-border-hover', theme.borderHover);
  root.style.setProperty('--theme-hero-gradient', theme.heroGradient);
  root.style.setProperty('--theme-sidebar-gradient', theme.sidebarGradient);
  root.style.setProperty('--theme-glow-color', theme.glowColor);
  root.style.setProperty('--theme-glow-intensity', theme.glowIntensity);
  root.style.setProperty('--theme-nav-active', theme.navActive);
  root.style.setProperty('--theme-nav-active-text', theme.navActiveText);
  root.style.setProperty('--theme-nav-active-border', theme.navActiveBorder);
  root.style.setProperty('--theme-live-indicator', theme.liveIndicator);
  root.style.setProperty('--theme-font', theme.font);

  // Apply background to body
  document.body.style.backgroundColor = theme.bg1;
  document.body.style.color = theme.textPrimary;
  document.body.style.fontFamily = theme.font;

  // Persist
  localStorage.setItem('dylandos:active-theme', theme.id);
};
```

```typescript
export const useSettings = () => {
  // ... existing ...

  const setTheme = useCallback((themeId: string) => {
    const theme = THEMES[themeId];
    if (!theme) return;
    applyTheme(theme);
    updateSettings({ theme: themeId, accentColor: theme.accent1 });
  }, [updateSettings]);

  // Apply theme on load
  useEffect(() => {
    const savedThemeId = settings.theme ?? 'default';
    const theme = THEMES[savedThemeId] ?? THEMES.default;
    applyTheme(theme);
  }, []);

  return { settings, updateSettings, setTheme };
```
